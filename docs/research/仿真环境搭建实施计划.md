## 三、仿真脚本

### 3.1 遥测模拟器 (`simulation/scripts/telemetry_simulator.py`)

```python
#!/usr/bin/env python3
"""
模拟无人机沿机场跑道盘旋飞行，每秒推送 MQTT 遥测数据
数据格式兼容 DJI Cloud API OSD 物模型
"""

import paho.mqtt.client as mqtt
import json
import time
import math

# ========== 配置 ==========
MQTT_BROKER = "localhost"
MQTT_PORT = 1883
TOPIC_PREFIX = "thing/product/DRONE001"

# 机场跑道参数（匹配现有 AirportData: 3000m x 50m）
RUNWAY_CENTER_LAT = 30.2500
RUNWAY_CENTER_LON = 120.1500
RUNWAY_LENGTH = 3000  # 米
RUNWAY_HEADING = 0    # 度（正北）

# 飞行参数
ALTITUDE = 100        # 米
SPEED = 15            # m/s
LOOP_DURATION = 180   # 秒（完成一圈盘旋的时间）

# ========== 视频流 URL（供指挥席拉流） ==========
VISIBLE_RTSP = "rtsp://localhost:8554/visible"
THERMAL_RTSP = "rtsp://localhost:8554/thermal"

client = mqtt.Client()
client.connect(MQTT_BROKER, MQTT_PORT, 60)

t = 0
while True:
    # 计算盘旋位置：椭圆轨迹覆盖跑道
    phase = (t % LOOP_DURATION) / LOOP_DURATION * 2 * math.pi

    lat_offset = (RUNWAY_LENGTH / 2 / 111320.0) * math.cos(phase) * 0.8
    lon_offset = (RUNWAY_LENGTH / 2 / (111320.0 * math.cos(math.radians(RUNWAY_CENTER_LAT)))) * math.sin(phase) * 0.3

    # 模拟航向（切线方向）
    heading = math.degrees(math.atan2(
        -math.sin(phase),
        -math.cos(phase) * 0.8
    )) % 360

    osd = {
        "latitude": RUNWAY_CENTER_LAT + lat_offset,
        "longitude": RUNWAY_CENTER_LON + lon_offset,
        "altitude": ALTITUDE + 5 * math.sin(phase * 3),  # 轻微高度波动
        "attitude_head": heading,
        "attitude_pitch": 2.0 * math.sin(phase * 2),
        "attitude_roll": 1.5 * math.sin(phase * 3),
        "horizontal_speed": SPEED,
        "vertical_speed": 0.0,
        "battery": {
            "value": max(20, 95 - (t % 3600) / 3600 * 75),
            "temperature": 35.0 + 2 * math.sin(phase)
        },
        "gps_number": 15,
        "rtk_status": "FIX",
        "flight_mode": "AUTO_MISSION",
        "live_stream": {
            "visible": VISIBLE_RTSP,
            "thermal": THERMAL_RTSP
        }
    }

    client.publish(f"{TOPIC_PREFIX}/osd", json.dumps(osd), qos=1)
    print(f"[{time.strftime('%H:%M:%S')}] OSD: lat={osd['latitude']:.6f} lon={osd['longitude']:.6f} alt={osd['altitude']:.1f}m heading={heading:.0f}°")

    t += 1
    time.sleep(1)
```

### 3.2 一键启动脚本 (`simulation/scripts/start_simulation.sh`)

```bash
#!/bin/bash
# 启动完整仿真环境：RTSP 服务器 + MQTT Broker + 视频推流 + 遥测模拟

set -e
SIM_DIR="$(cd "$(dirname "$0")/.." && pwd)"
echo "=== UXO 仿真环境启动 ==="

# 1. 启动 MQTT Broker（如果未运行）
if ! pgrep -x mosquitto > /dev/null; then
    echo "[1/5] 启动 MQTT Broker..."
    mosquitto -d
else
    echo "[1/5] MQTT Broker 已在运行"
fi

# 2. 启动 RTSP 服务器
if ! pgrep -x mediamtx > /dev/null; then
    echo "[2/5] 启动 RTSP 服务器 (MediaMTX)..."
    mediamtx &
    sleep 1
else
    echo "[2/5] RTSP 服务器已在运行"
fi

# 3. 推送可见光视频流
echo "[3/5] 推送可见光视频流 → rtsp://localhost:8554/visible"
ffmpeg -re -stream_loop -1 \
    -i "$SIM_DIR/videos/visible_1.mp4" \
    -c:v copy -an \
    -f rtsp rtsp://localhost:8554/visible &
VISIBLE_PID=$!

# 4. 推送红外视频流
echo "[4/5] 推送红外视频流 → rtsp://localhost:8554/thermal"
ffmpeg -re -stream_loop -1 \
    -i "$SIM_DIR/videos/thermal_1.mp4" \
    -c:v copy -an \
    -f rtsp rtsp://localhost:8554/thermal &
THERMAL_PID=$!

sleep 2

# 5. 启动遥测模拟器
echo "[5/5] 启动 MQTT 遥测模拟器..."
python3 "$SIM_DIR/scripts/telemetry_simulator.py" &
TELEM_PID=$!

echo ""
echo "=== 仿真环境已启动 ==="
echo "可见光 RTSP: rtsp://localhost:8554/visible"
echo "红外 RTSP:   rtsp://localhost:8554/thermal"
echo "MQTT Broker: localhost:1883"
echo "MQTT Topic:  thing/product/DRONE001/osd"
echo ""
echo "进程 PID:"
echo "  mediamtx:    $(pgrep mediamtx)"
echo "  mosquitto:   $(pgrep mosquitto)"
echo "  ffmpeg(vis): $VISIBLE_PID"
echo "  ffmpeg(thm): $THERMAL_PID"
echo "  telemetry:   $TELEM_PID"
echo ""
echo "按 Ctrl+C 停止所有仿真进程"

# 等待 Ctrl+C
trap "kill $VISIBLE_PID $THERMAL_PID $TELEM_PID 2>/dev/null; echo '仿真环境已停止'" EXIT
wait
```

```cpp
#pragma once
#include <QObject>
#include <QImage>
#include <QString>

namespace Core {
namespace VideoStream {

/**
 * @brief RTSP 视频流接收器抽象接口
 *
 * 仿真阶段：连接本地 rtsp://localhost:8554/visible
 * 真机阶段：连接无人机实际推流地址
 * 切换时仅改变 URL，其余代码不变
 */
class IRtspReceiver : public QObject {
    Q_OBJECT
public:
    explicit IRtspReceiver(QObject* parent = nullptr) : QObject(parent) {}
    virtual ~IRtspReceiver() = default;

    /// 连接到指定 RTSP URL
    virtual bool open(const QString& url) = 0;

    /// 获取最新一帧（返回 QImage，用于 UI 渲染）
    virtual QImage currentFrame() const = 0;

    /// 获取原始帧数据（用于 OpenCV AI 推理，可后续扩展）
    virtual const uint8_t* rawData() const = 0;
    virtual int width() const = 0;
    virtual int height() const = 0;

    /// 断开连接
    virtual void close() = 0;

    /// 是否已连接
    virtual bool isOpen() const = 0;

signals:
    /// 新帧到达信号，驱动 UI 刷新
    void frameReady(const QImage& frame);

    /// 连接状态变化
    void connectionStateChanged(bool connected, const QString& errorMsg);
};

} // namespace VideoStream
} // namespace Core
```

**FFmpeg 实现**（`src/Core/VideoStream/FFmpegReceiver.cpp`）：

```cpp
#include "Core/VideoStream/FFmpegReceiver.h"
#include <QThread>
#include <QDateTime>

extern "C" {
#include <libavformat/avformat.h>
#include <libavcodec/avcodec.h>
#include <libswscale/swscale.h>
#include <libavutil/imgutils.h>
}

namespace Core {
namespace VideoStream {

class FFmpegReceiver::Private : public QThread {
    Q_OBJECT
public:
    Private(FFmpegReceiver* q) : q_ptr(q) {}

    void run() override {
        // 1. 打开 RTSP 流
        avformat_open_input(&fmtCtx, url.toUtf8().constData(), nullptr, nullptr);
        avformat_find_stream_info(fmtCtx, nullptr);

        // 2. 查找视频流 + 打开解码器
        int videoIdx = av_find_best_stream(fmtCtx, AVMEDIA_TYPE_VIDEO, -1, -1, nullptr, 0);
        auto* params = fmtCtx->streams[videoIdx]->codecpar;
        auto* codec = avcodec_find_decoder(params->codec_id);
        codecCtx = avcodec_alloc_context3(codec);
        avcodec_parameters_to_context(codecCtx, params);
        avcodec_open2(codecCtx, codec, nullptr);

        // 3. 初始化转换器 (YUV → RGB)
        swsCtx = sws_getContext(params->width, params->height, (AVPixelFormat)params->format,
                                params->width, params->height, AV_PIX_FMT_RGB32,
                                SWS_BILINEAR, nullptr, nullptr, nullptr);

        frameW = params->width;
        frameH = params->height;

        // 4. 逐帧解码循环
        AVPacket* pkt = av_packet_alloc();
        AVFrame* frame = av_frame_alloc();
        AVFrame* rgbFrame = av_frame_alloc();
        rgbFrame->format = AV_PIX_FMT_RGB32;
        rgbFrame->width = frameW;
        rgbFrame->height = frameH;
        av_frame_get_buffer(rgbFrame, 0);

        while (!isInterruptionRequested()) {
            if (av_read_frame(fmtCtx, pkt) < 0) break;
            if (pkt->stream_index != videoIdx) { av_packet_unref(pkt); continue; }

            avcodec_send_packet(codecCtx, pkt);
            while (avcodec_receive_frame(codecCtx, frame) == 0) {
                sws_scale(swsCtx, frame->data, frame->linesize, 0, frameH,
                          rgbFrame->data, rgbFrame->linesize);

                // 构造 QImage（深拷贝，线程安全）
                QImage img(rgbFrame->data[0], frameW, frameH, QImage::Format_RGB32);
                emit q_ptr->frameReady(img.copy());
            }
            av_packet_unref(pkt);
        }

        // 清理
        av_frame_free(&rgbFrame);
        av_frame_free(&frame);
        av_packet_free(&pkt);
        avcodec_free_context(&codecCtx);
        avformat_close_input(&fmtCtx);
    }

    void stop() {
        requestInterruption();
        wait(3000);
    }

    QString url;
    AVFormatContext* fmtCtx = nullptr;
    AVCodecContext* codecCtx = nullptr;
    SwsContext* swsCtx = nullptr;
    int frameW = 0, frameH = 0;

private:
    FFmpegReceiver* q_ptr;
};

// === Public API ===
FFmpegReceiver::FFmpegReceiver(QObject* parent) : IRtspReceiver(parent)
    , d(new Private(this)) {}

FFmpegReceiver::~FFmpegReceiver() { close(); }

bool FFmpegReceiver::open(const QString& url) {
    d->url = url;
    d->start();
    return true;
}

void FFmpegReceiver::close() {
    d->stop();
}

QImage FFmpegReceiver::currentFrame() const {
    return m_currentFrame;
}

bool FFmpegReceiver::isOpen() const {
    return d->isRunning();
}

// ... 其余接口实现
} // namespace VideoStream
} // namespace Core
```

## 八、风险与备用方案

| 风险 | 备用方案 |
|------|---------|
| FFmpeg 解码线程开销大 | 使用 QMediaPlayer（Qt内置，支持 RTSP）作为轻量替代 |
| QMqttClient 不可用 | 改用 mosquitto 的 C 客户端库 (`libmosquitto-dev`) |
| WSL 中 RTSP 端口无法访问 | 在 Windows 侧安装 MediaMTX 和 Mosquitto 的 Windows 版本 |
| 红外视频素材难以获取 | 用可见光视频 + OpenCV applyColorMap 生成伪彩色模拟 |
